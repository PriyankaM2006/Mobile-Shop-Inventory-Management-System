JavaScript Implementation

function payNow(product){
    var url = "/add_mobile?type=" + encodeURIComponent(product.type)
        + "&model=" + encodeURIComponent(product.model)
        + "&price=" + product.price;
    if (product.ram) url += "&ram=" + encodeURIComponent(product.ram);
    if (product.rom) url += "&rom=" + encodeURIComponent(product.rom);
    if (product.capacity) url += "&capacity=" + encodeURIComponent(product.capacity);
    window.location.href = url;
}
